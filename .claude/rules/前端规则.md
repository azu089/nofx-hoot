## TradingView K线图

### 轻量级图表

```tsx
'use client';
import { useEffect, useRef } from 'react';
import { createChart, IChartApi } from 'lightweight-charts';

export function TradingChart({ data }: { data: any[] }) {
  const chartRef = useRef<HTMLDivElement>(null);
  const chart = useRef<IChartApi | null>(null);

  useEffect(() => {
    if (!chartRef.current) return;

    chart.current = createChart(chartRef.current, {
      width: chartRef.current.clientWidth,
      height: 400,
      layout: {
        background: { color: '#0B0E11' },
        textColor: '#848E9C',
      },
      grid: {
        vertLines: { color: '#2B3139' },
        horzLines: { color: '#2B3139' },
      },
    });

    const candlestickSeries = chart.current.addCandlestickSeries({
      upColor: '#00C087',
      downColor: '#F23645',
    });
    candlestickSeries.setData(data);

    return () => chart.current?.remove();
  }, [data]);

  return <div ref={chartRef} className="w-full" />;
}
```

## TanStack Query 数据请求

### 查询

```tsx
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { api } from '@/lib/api';

// 查询
export function useDashboard() {
  return useQuery({
    queryKey: ['dashboard'],
    queryFn: () => api.get('/dashboard'),
  });
}

// 变更
export function useDeposit() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: (data: { amount: string }) => api.post('/wallet/deposit', data),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['wallet'] });
    },
  });
}
```

### 页面使用

```tsx
export default function DashboardPage() {
  const { data, isLoading, error } = useDashboard();

  if (isLoading) return <DashboardSkeleton />;
  if (error) return <ErrorState onRetry={() => refetch()} />;

  return <DashboardContent data={data} />;
}
```

## 页面状态规范

每个页面必须处理 4 种状态：

| 状态 | 说明 | 组件 |
|------|------|------|
| Loading | 加载中 | Skeleton / Spinner |
| Empty | 空数据 | EmptyState |
| Error | 错误 | ErrorState + 重试 |
| Success | 成功 | 正常内容 |

```tsx
function Page() {
  const { data, isLoading, error, refetch } = useQuery(...);

  if (isLoading) return <PageSkeleton />;
  if (error) return <ErrorState onRetry={refetch} />;
  if (!data?.length) return <EmptyState />;

  return <PageContent data={data} />;
}
```

## 安全规范

1. **禁止在前端计算资金**（必须后端计算）
2. **禁止存储敏感信息到 localStorage**
3. **使用环境变量配置 API URL**
4. **Web3 签名前必须展示清晰的确认信息**
